### 力矩计算细节

```java
// 对每个活水车，遍历其 4 个相邻槽位
for (int neighbor : getNeighbors(wheelSlot, containerSize, width)) {
    FlowEntry fe = flows.get(neighbor);
    if (fe == null) continue;           // 无水流
    if (fe.fromSlot < 0) continue;      // 水源无方向

    // 水流方向
    int fdx = (neighbor % width) - (fe.fromSlot % width);
    int fdy = (neighbor / width) - (fe.fromSlot / width);

    // 位置向量（水车中心 → 相邻槽位）
    int px = (neighbor % width) - (wheelSlot % width);
    int py = (neighbor / width) - (wheelSlot / width);

    // 二维叉积 = 力矩
    int torque = px * fdy - py * fdx;

    // 水流强度乘数（越靠近水源越强）
    float strength = (MAX_FLOW_LEVEL - fe.level + 1.0f) / MAX_FLOW_LEVEL;

    // 累加力矩
    netTorque += torque * strength * stackSize;
}

// netTorque > 0 → CW, < 0 → CCW
```
